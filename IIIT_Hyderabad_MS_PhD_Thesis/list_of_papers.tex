The following are the key papers that the report \emph{Tabular Data Understanding} draws upon and that form the basis for discussion, organized by the two pillars around which the report is structured.

\vspace*{0.3cm}
\noindent\textbf{Foundations and Representations}
\begin{enumerate}[label={[S\arabic*]}]
    \item X. Wu, A. Ritter, and W. Xu, \textbf{``Tabular Data Understanding with LLMs: A Survey of Recent Advances and Challenges"}, {\it arXiv:2508.00217}, 2025.

    \item Y. Sui, M. Zhou, M. Zhou, S. Han, and D. Zhang, \textbf{``Table Meets LLM: Can Large Language Models Understand Structured Table Data? A Benchmark and Empirical Study"}, in {\it Proceedings of the 17th ACM International Conference on Web Search and Data Mining (WSDM)}, 2024.
\end{enumerate}

\vspace*{0.2cm}
\noindent\textbf{Schema Understanding (Text-to-SQL)}
\begin{enumerate}[start=3,label={[S\arabic*]}]
    \item T. Yu, R. Zhang, K. Yang, M. Yasunaga, D. Wang, et al., \textbf{``Spider: A Large-Scale Human-Labeled Dataset for Complex and Cross-Domain Semantic Parsing and Text-to-SQL Task"}, in {\it Proceedings of EMNLP}, 2018.

    \item J. Li, B. Hui, G. Qu, J. Yang, B. Li, et al., \textbf{``Can LLM Already Serve as a Database Interface? A Big Bench for Large-Scale Database Grounded Text-to-SQLs (BIRD)"}, in {\it Advances in Neural Information Processing Systems (NeurIPS)}, 2023.

    \item M. Pourreza, H. Li, R. Sun, Y. Chung, S. Talaei, et al., \textbf{``CHASE-SQL: Multi-Path Reasoning and Preference Optimized Candidate Selection in Text-to-SQL"}, {\it arXiv:2410.01943}, 2024.

    \item A. Asai, Z. Wu, Y. Wang, A. Sil, and H. Hajishirzi, \textbf{``Self-RAG: Learning to Retrieve, Generate, and Critique through Self-Reflection"}, {\it arXiv:2310.11511}, 2023.
\end{enumerate}

\vspace*{0.2cm}
\noindent\textbf{Data Understanding}
\begin{enumerate}[start=7,label={[S\arabic*]}]
    \item P. Pasupat and P. Liang, \textbf{``Compositional Semantic Parsing on Semi-Structured Tables"}, in {\it Proceedings of the 53rd Annual Meeting of the Association for Computational Linguistics (ACL)}, 2015.

    \item Z. Cheng, H. Dong, Z. Wang, R. Jia, J. Guo, et al., \textbf{``HiTab: A Hierarchical Table Dataset for Question Answering and Natural Language Generation"}, in {\it Proceedings of ACL}, 2022.

    \item W. Chen, H. Wang, J. Chen, Y. Zhang, H. Wang, et al., \textbf{``TabFact: A Large-Scale Dataset for Table-Based Fact Verification"}, in {\it International Conference on Learning Representations (ICLR)}, 2020.
\end{enumerate}

\vspace*{0.2cm}
\noindent\textbf{Graph Representation of Tabular Data}
\begin{enumerate}[start=10,label={[S\arabic*]}]
    \item Q. Li, C. Huang, S. Li, Y. Xiang, D. Xiong, and W. Lei, \textbf{``GraphOTTER: Evolving LLM-Based Graph Reasoning for Complex Table Question Answering"}, in {\it Proceedings of the 31st International Conference on Computational Linguistics (COLING)}, 2025.

    \item X. He, Y. Tian, Y. Sun, N. V. Chawla, T. Laurent, Y. LeCun, X. Bresson, and B. Hooi, \textbf{``G-Retriever: Retrieval-Augmented Generation for Textual Graph Understanding and Question Answering"}, in {\it Advances in Neural Information Processing Systems (NeurIPS)}, 2024.

    \item D. Edge, H. Trinh, N. Cheng, J. Bradley, A. Chao, et al., \textbf{``From Local to Global: A Graph RAG Approach to Query-Focused Summarization"}, {\it arXiv:2404.16130}, 2024.
\end{enumerate}
